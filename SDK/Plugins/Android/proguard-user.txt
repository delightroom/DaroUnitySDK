# Daro Unity SDK — supplemental R8 / ProGuard keep rules.
#
# Unity 6 automatically merges any `proguard-user.txt` placed under
# Plugins/Android/ into the consumer's Gradle build. The Editor postprocessor
# mirrors these rules into exported Gradle projects as a fallback because some
# Unity exports omit package-level proguard-user.txt inputs.
#
# Shim keeps (`so.daro.unity.*`) live inside daro-android-wrapper.aar's
# embedded `proguard.txt` (consumerProguardFiles in build.gradle.kts) —
# they're applied automatically when consumer Gradle includes the AAR.
# This file only carries keeps that the AAR cannot embed:

# ── Native model classes accessed via JNI field reads ───────────────────────
# The shim's listener implementations read `err.code`, `err.message`,
# `err.latency`, `error.message`, `adInfo.latency`, `rewardItem.type`,
# `rewardItem.amount`. R8 may rename these getters under aggressive minify
# settings; pin them. These types belong to daro-m, not the shim, so they
# can't ride in the shim's AAR proguard.txt.

-keep class droom.daro.core.model.DaroAdLoadError { *; }
-keep class droom.daro.core.model.DaroAdDisplayFailError { *; }
-keep class droom.daro.core.model.DaroAdInfo { *; }
-keep class droom.daro.core.model.DaroRewardedAd$DaroRewardedItem { *; }

# ── Native ad publisher-rendered bridge internals ───────────────────────────
# Android native ads use a Unity-rendered CTA overlay and reflect into daro-m's
# DaroAdView public lifecycle setters to disable its default auto reload path.
# Keep the reflected methods stable under consumer minification.

-keepclassmembers class droom.daro.view.DaroAdView {
    public void setAutoDetectLifecycle(boolean);
    public void setRefreshSeconds(int);
}
